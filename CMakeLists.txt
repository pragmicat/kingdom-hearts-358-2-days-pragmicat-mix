cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" PRAGMICAT_VERSION LIMIT_COUNT 1)
project(PragmicatMix VERSION ${PRAGMICAT_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(pragmicat_mix MODULE src/combat.cpp src/limit_finishers.cpp
  src/combat_motion.cpp src/combat_rules.cpp src/drive_form.cpp src/animation_links.cpp src/drive_hud.cpp
  src/practice_boot.cpp src/weapon_animation.cpp)
target_include_directories(pragmicat_mix PRIVATE vendor/sdk/include generated)
set_target_properties(pragmicat_mix PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  target_compile_options(pragmicat_mix PRIVATE /W4 /EHsc /utf-8)
else()
  target_compile_options(pragmicat_mix PRIVATE -Wall -Wextra)
endif()
if(WIN32)
  enable_language(RC)
  file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/version.h" CONTENT
    "#define PRAGMICAT_VERSION_NUM ${PROJECT_VERSION_MAJOR},${PROJECT_VERSION_MINOR},${PROJECT_VERSION_PATCH},0\n#define PRAGMICAT_VERSION_STRING \"${PROJECT_VERSION}\"\n" NEWLINE_STYLE UNIX)
  target_sources(pragmicat_mix PRIVATE launcher/resources.rc)
  target_include_directories(pragmicat_mix PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" assets)
  target_compile_definitions(pragmicat_mix PRIVATE PRAGMICAT_PLUGIN)
  target_link_libraries(pragmicat_mix PRIVATE user32 gdi32)
endif()
